def format_phone_number(phone_number):
    """Format phone number to show only last 10 digits"""
    cleaned = re.sub(r'\D', '', str(phone_number))
    if cleaned.startswith('91') and len(cleaned) > 10:
        return cleaned[-10:]
    if len(cleaned) > 10:
        return cleaned[-10:]
    return cleaned

def get_speedz_headers():

        "Content-Type": "application/json"

def speedz_check_number(phone_number, retry_count=0):
    """Check if a number is registered using SpeedzChecker API"""
    global last_api_call, speedz_api_working
    
    api_number = phone_number
    if not phone_number.startswith('91') and len(phone_number) == 10:
        api_number = '91' + phone_number
    
    try:
        current_time = time.time()
        time_since_last = current_time - last_api_call
        if time_since_last < 5:
            time.sleep(5 - time_since_last)
        
        url = f"{SPEEDZ_BASE_URL}/check"
        payload = {
            "service": "meesho",
            "number": api_number
        }
        
        print(f"🔍 Checking {api_number} with SpeedzChecker API...")
        response = requests.post(url, json=payload, headers=get_speedz_headers(), timeout=20)
        last_api_call = time.time()
        
        if response.status_code == 200:
            data = response.json()
            print(f"✅ SpeedzChecker response: {data}")
            speedz_api_working = True
            
            if isinstance(data, dict):
                if data.get('success') == True:
                    is_registered = data.get('is_registered', False)
                    is_down = data.get('is_down', False)
                    
                    if is_down:
                        return "error"
                    
                    if is_registered:
                        return "registered"
                    else:
                        return "not_registered"
            
            return "not_registered"
        
        elif response.status_code == 401:
            print(f"❌ SpeedzChecker API - Invalid/Expired API Key (401)")
            speedz_api_working = False
            return "api_key_error"
        
        elif response.status_code == 429:
            print(f"❌ SpeedzChecker API - Rate Limit (429)")
            return "rate_limit"
        
        print(f"❌ SpeedzChecker API error: {response.status_code}")
        return "error"
        
    except Exception as e:
        print(f"❌ SpeedzChecker API exception: {e}")
        if retry_count < 2:
            print(f"🔄 Retrying... ({retry_count + 1}/2)")
            time.sleep(3)
            return speedz_check_number(phone_number, retry_count + 1)
        return "error"

# ========== OTP DOCTOR FUNCTIONS ==========

def check_balance():
    """Check OTPDoctor account balance"""
    url = f"{OTP_BASE_URL}?action=getBalance&api_key={OTP_API_KEY}"
    try:
        response = requests.get(url, timeout=10)
        if response.status_code == 200:
            data = response.text
            if "ACCESS_BALANCE" in data:
                balance = data.split(":")[1]
                return float(balance)
        return None
    except Exception as e:
        print(f"Balance check error: {e}")
        return None

def purchase_number(service_id, max_price=None):
    """Purchase a number from OTPDoctor"""
    url = f"{OTP_BASE_URL}?action=getNumber&api_key={OTP_API_KEY}&service={service_id}"
    if max_price:
        url += f"&maxPrice={max_price}"
    
    try:
        response = requests.get(url, timeout=15)
        if response.status_code == 200:
            data = response.text
            if "ACCESS_NUMBER" in data:
                parts = data.split(":")
                activation_id = parts[1]
                phone_number = parts[2]
                
                service_info = MEESHO_SERVICES.get(service_id, {})
                price = service_info.get("price", "Unknown")
                server = service_info.get("server", "Unknown")
                
                return {
                    "success": True,
                    "activation_id": activation_id,
                    "phone_number": phone_number,
                    "price": price,
                    "server": server,
                    "service_name": service_info.get("name", "Unknown")
                }
            elif "NO_NUMBERS" in data:
                return {"success": False, "error": "No numbers available"}
            elif "NO_BALANCE" in data:
                return {"success": False, "error": "Insufficient balance"}
            elif "BAD_KEY" in data:
                return {"success": False, "error": "Invalid API key"}
            elif "BAD_SERVICE" in data:
                return {"success": False, "error": "Bad service ID"}
        return {"success": False, "error": "Unknown error"}
    except Exception as e:
        return {"success": False, "error": str(e)}

def check_activation_status(activation_id):
    """Check activation status and get SMS"""
    url = f"{OTP_BASE_URL}?action=getStatus&api_key={OTP_API_KEY}&id={activation_id}"
    try:
        response = requests.get(url, timeout=10)
        if response.status_code == 200:
            data = response.text
            if "STATUS_OK" in data:
                sms_text = data.split(":", 1)[1] if ":" in data else ""
                return {"status": "OK", "sms": sms_text}
            elif "STATUS_WAIT_CODE" in data:
                return {"status": "WAITING"}
            elif "STATUS_CANCEL" in data:
                return {"status": "CANCELLED"}
            elif "NO_ACTIVATION" in data:
                return {"status": "NOT_FOUND"}
            elif "BAD_KEY" in data:
                return {"status": "INVALID_API_KEY"}
        return {"status": "ERROR"}
    except Exception as e:
        return {"status": "ERROR", "error": str(e)}

def cancel_activation(activation_id):
    """Cancel an activation on OTPDoctor"""
    url = f"{OTP_BASE_URL}?action=setStatus&api_key={OTP_API_KEY}&id={activation_id}&status=8"
    try:
        print(f"🔄 Cancelling activation {activation_id}...")
        response = requests.get(url, timeout=10)
        
        if response.status_code == 200:
            data = response.text
            print(f"📝 Cancel response: {data}")
            
            if "STATUS_CANCEL" in data:
                print(f"✅ Successfully cancelled activation {activation_id}")
                return {"success": True, "message": data}
            elif "ACCESS_ACTIVATION" in data:
                print(f"ℹ️ Activation already cancelled/completed")
                return {"success": True, "message": "Already cancelled"}
            else:
                print(f"❌ Failed to cancel: {data}")
                return {"success": False, "error": data}
        else:
            print(f"❌ HTTP Error: {response.status_code}")
            return {"success": False, "error": f"HTTP {response.status_code}"}
            
    except Exception as e:
        print(f"❌ Cancel exception: {e}")
        return {"success": False, "error": str(e)}

async def send_telegram_message(text, chat_id=None, reply_markup=None):
    """Send message to Telegram with optional inline keyboard"""
    if chat_id is None:
        chat_id = TELEGRAM_CHAT_ID
    
    try:
        url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
        payload = {
            "chat_id": chat_id,
            "text": text,
            "parse_mode": "HTML"
        }
        if reply_markup:
            payload["reply_markup"] = reply_markup.to_dict()
        
        response = requests.post(url, json=payload, timeout=10)
        return response.status_code == 200
    except Exception as e:
        print(f"Telegram send error: {e}")
        return False

async def fetch_otp(activation_id, phone_number, price):
    """Fetch OTP for the number"""
    attempts = 0
    max_attempts = 40
    
    print(f"🔄 Starting OTP fetch for {phone_number}...")
    
    keyboard = [
        [InlineKeyboardButton("❌ Cancel Number", callback_data=f"cancel_{activation_id}")],
        [InlineKeyboardButton("🔄 Request Next SMS", callback_data=f"nextsms_{activation_id}")]
    ]
    reply_markup = InlineKeyboardMarkup(keyboard)
    
    await send_telegram_message(
        f"🔄 <b>Starting OTP fetch</b>\n"
        f"📱 Number: <code>{phone_number}</code>\n"
        f"🆔 Activation: <code>{activation_id}</code>\n"
        f"💰 Amount Debited: ₹{price}\n"
        f"⏱️ Waiting for OTP... (4 minutes timeout)",
        reply_markup=reply_markup
    )
    
    while attempts < max_attempts:
        try:
            status = check_activation_status(activation_id)
            
            if status.get("status") == "INVALID_API_KEY":
                await send_telegram_message(
                    f"❌ <b>Invalid OTPDoctor API Key!</b>\n"
                    f"Please update your API key."
                )
                return False
            
            if status.get("status") == "OK":
                sms = status.get("sms", "")
                otp_pattern = r'\b\d{4,6}\b'
                otp_matches = re.findall(otp_pattern, sms)
                
                otp = otp_matches[0] if otp_matches else "Unable to extract OTP"
                
                print(f"✅ OTP received for {phone_number}: {otp}")
                
                keyboard = [
                    [InlineKeyboardButton("❌ Cancel Number", callback_data=f"cancel_{activation_id}")],
                    [InlineKeyboardButton("✅ Done (Cancel)", callback_data=f"done_{activation_id}")]
                ]
                reply_markup = InlineKeyboardMarkup(keyboard)
                
                await send_telegram_message(
                    f"🔐 <b>✅ OTP RECEIVED!</b>\n\n"
                    f"📱 Number: <code>{phone_number}</code>\n"
                    f"🆔 Activation: <code>{activation_id}</code>\n"
                    f"💰 Amount Debited: ₹{price}\n"
                    f"🔑 OTP: <code>{otp}</code>\n"
                    f"📝 Full SMS: {sms}\n"
                    f"⏰ Time: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n\n"
                    f"⚠️ Auto-cancelling in 60 seconds...",
                    reply_markup=reply_markup
                )
                
                await asyncio.sleep(60)
                force_cancel_activation(activation_id)
                await send_telegram_message(
                    f"⏹️ <b>Activation Cancelled</b>\n"
                    f"📱 Number: <code>{phone_number}</code>\n"
                    f"🆔 Activation: <code>{activation_id}</code>\n"
                    f"💰 Amount Debited: ₹{price}\n"
                    f"🔑 OTP: <code>{otp}</code>"
                )
                print(f"⏹️ Cancelled activation for {phone_number}")
                return True
                
            elif status.get("status") == "CANCELLED":
                await send_telegram_message(
                    f"⏹️ <b>Activation Cancelled</b>\n"
                    f"📱 Number: <code>{phone_number}</code>\n"
                    f"🆔 Activation: <code>{activation_id}</code>\n"
                    f"💰 Amount Debited: ₹{price}"
                )
                print(f"⏹️ Activation cancelled for {phone_number}")
                return False
                
            elif status.get("status") == "ERROR":
                print(f"⚠️ Error checking status for {phone_number}")
                
            await asyncio.sleep(6)
            attempts += 1
            
            if attempts % 10 == 0:
                remaining = (max_attempts - attempts) * 6 // 60
                keyboard = [
                    [InlineKeyboardButton("❌ Cancel Number", callback_data=f"cancel_{activation_id}")]
                ]
                reply_markup = InlineKeyboardMarkup(keyboard)
                await send_telegram_message(
                    f"⏳ Still waiting for OTP... ({remaining} min remaining)\n"
                    f"📱 Number: <code>{phone_number}</code>\n"
                    f"🆔 Activation: <code>{activation_id}</code>\n"
                    f"💰 Amount Debited: ₹{price}",
                    reply_markup=reply_markup
                )
            
        except Exception as e:
            print(f"OTP fetch error for {phone_number}: {e}")
            await asyncio.sleep(6)
            attempts += 1
    
    force_cancel_activation(activation_id)
    await send_telegram_message(
        f"⏰ <b>⏱️ Timeout - No OTP Received</b>\n"
        f"📱 Number: <code>{phone_number}</code>\n"
        f"🆔 Activation: <code>{activation_id}</code>\n"
        f"💰 Amount Debited: ₹{price}\n"
        f"Activation cancelled after 4 minutes"
    )
    print(f"⏰ Timeout for {phone_number} - Cancelled")
    return False

async def purchase_loop():
    """Main loop to continuously purchase numbers"""
    global is_running, total_purchased, unregistered_found, speedz_api_working
    
    service_ids = list(MEESHO_SERVICES.keys())
    service_index = 0
    
    print("\n" + "="*60)
    print("🚀 Starting Continuous Purchase Loop")
    print("="*60)
    print(f"📱 Bot: @Meeshonumbot")
    print(f"📋 Services: {len(MEESHO_SERVICES)}")
    print(f"🔗 SpeedzChecker API: {'✅ Connected' if speedz_api_working else '❌ Failed'}")
    print("="*60 + "\n")
    
    await send_telegram_message(
        f"🚀 <b>Bot Started - Continuous Purchasing!</b>\n"
        f"📱 @Meeshonumbot is now active\n"
        f"📋 Monitoring {len(MEESHO_SERVICES)} services\n"
        f"🔗 SpeedzChecker API: {'✅ Connected' if speedz_api_working else '❌ Failed'}\n"
        f"🔄 Buying numbers continuously..."
    )
    
    while is_running:
        try:
            # Check SpeedzChecker API status
            if not speedz_api_working:
                print("⏳ SpeedzChecker API not working, waiting...")
                await asyncio.sleep(30)
                test_result = speedz_check_number("9876543210")
                if test_result != "api_key_error":
                    speedz_api_working = True
                    await send_telegram_message(
                        f"✅ <b>SpeedzChecker API is back online!</b>"
                    )
                continue
            
            # Check for pending error numbers to retry
            if error_numbers:
                print(f"🔄 Found {len(error_numbers)} error numbers to retry")
                for activation_id in list(error_numbers.keys()):
                    if error_numbers[activation_id]['retries'] < 3:
                        asyncio.create_task(retry_error_number(activation_id))
                        await asyncio.sleep(5)
            
            balance = check_balance()
            if balance is None:
                print("⚠️ Could not fetch balance")
                await asyncio.sleep(5)
                continue
                
            if balance < 5:
                print(f"⚠️ Low balance: ₹{balance}")
                await send_telegram_message(
                    f"⚠️ <b>BALANCE ALERT</b>\n"
                    f"Current Balance: ₹{balance}\n"
                    f"Minimum required: ₹5\n"
                    f"Pausing purchases..."
                )
                await asyncio.sleep(30)
                continue
            
            purchase_success = False
            
            for attempt in range(len(service_ids)):
                service_id = service_ids[service_index]
                service_info = MEESHO_SERVICES[service_id]
                service_name = service_info["name"]
                service_index = (service_index + 1) % len(service_ids)
                
                print(f"\n📱 Trying service {service_id} ({service_name})...")
                print(f"💰 Current balance: ₹{balance}")
                print(f"📊 Total Purchased: {total_purchased}")
                print(f"🎯 Unregistered Found: {unregistered_found}")
                
                result = purchase_number(service_id)
                
                if result.get("success"):
                    phone = result["phone_number"]
                    activation = result["activation_id"]
                    price = result["price"]
                    server = result["server"]
                    total_purchased += 1
                    purchase_success = True
                    
                    formatted_phone = format_phone_number(phone)
                    
                    print(f"✅ Purchased #{total_purchased}: {formatted_phone} (ID: {activation})")
                    print(f"💰 Price: ₹{price} | 🖥️ Server: {server}")
                    
                    asyncio.create_task(process_number(phone, activation, service_id, service_name, price, server))
                    break
                    
                elif "No numbers available" in result.get("error", ""):
                    print(f"❌ No numbers available for {service_name}, trying next...")
                    continue
                    
                elif "Invalid API key" in result.get("error", ""):
                    print("❌ Invalid OTPDoctor API key!")
                    await send_telegram_message(
                        "❌ <b>Invalid OTPDoctor API Key!</b>\n"
                        "Please update your API key."
                    )
                    await asyncio.sleep(30)
                    purchase_success = True
                    break
                    
                elif "Insufficient balance" in result.get("error", ""):
                    print("❌ Insufficient balance!")
                    await send_telegram_message(
                        "⚠️ <b>🚫 BALANCE ALERT</b>\n"
                        "Insufficient balance to purchase numbers!\n"
                        f"Current Balance: ₹{balance}\n"
                        "Please recharge your OTPDoctor account."
                    )
                    await asyncio.sleep(60)
                    purchase_success = True
                    break
                    
                else:
                    print(f"❌ Purchase error: {result}")
                    continue
            
            if not purchase_success:
                print("❌ No numbers available on any service, waiting 5 seconds...")
                await asyncio.sleep(5)
            else:
                await asyncio.sleep(2)
                
        except Exception as e:
            print(f"❌ Loop error: {e}")
            await asyncio.sleep(5)
    
    print("\n" + "="*60)
    print("⏹️ Purchase Loop Stopped")
    print(f"📊 Total Purchased: {total_purchased}")
    print(f"🎯 Unregistered Found: {unregistered_found}")
    print(f"💰 Total Refunded: ₹{total_refunded}")
    print(f"⚠️ Error Numbers: {len(error_numbers)}")
    print("="*60)
    
    await send_telegram_message(
        f"⏹️ <b>Bot Stopped</b>\n"
        f"📊 Total Purchased: {total_purchased}\n"
        f"🎯 Unregistered Found: {unregistered_found}\n"
        f"💰 Total Refunded: ₹{total_refunded}\n"
        f"⚠️ Error Numbers: {len(error_numbers)} (not cancelled)"
    )
